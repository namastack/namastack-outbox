SELECT
    "ID"
FROM "RUN"."OUTBOX_RECORD"
    /* RUN.IDX_OUTBOX_RECORD_STATUS_RETRY: STATUS = 'FAILED'
        AND NEXT_RETRY_AT < TIMESTAMP WITH TIME ZONE '2025-12-02 12:00:00+00'
     */
    /* scanCount: 1 */
WHERE ("STATUS" = 'FAILED')
    AND ("NEXT_RETRY_AT" < TIMESTAMP WITH TIME ZONE '2025-12-02 12:00:00+00')
ORDER BY "NEXT_RETRY_AT"
FETCH FIRST 1000 ROWS ONLY
