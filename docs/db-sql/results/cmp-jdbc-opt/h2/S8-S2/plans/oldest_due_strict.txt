SELECT
    "O"."NEXT_RETRY_AT"
FROM "RUN"."OUTBOX_RECORD" "O"
    /* RUN.IDX_OUTBOX_RECORD_STATUS_COMPLETED_CREATED_ID: NEXT_RETRY_AT <= TIMESTAMP WITH TIME ZONE '2026-01-01 12:00:00+00'
        AND COMPLETED_AT IS NULL
        AND STATUS = 'NEW'
     */
    /* scanCount: 1201 */
WHERE ("O"."PARTITION_NO" IN(0, 4, 8, 12, 16, 20, 24, 28, 32, 36, 40, 44, 48, 52, 56, 60, 64, 68, 72, 76, 80, 84, 88, 92, 96, 100, 104, 108, 112, 116, 120, 124, 128, 132, 136, 140, 144, 148, 152, 156, 160, 164, 168, 172, 176, 180, 184, 188, 192, 196, 200, 204, 208, 212, 216, 220, 224, 228, 232, 236, 240, 244, 248, 252))
    AND ("O"."NEXT_RETRY_AT" <= TIMESTAMP WITH TIME ZONE '2026-01-01 12:00:00+00')
    AND ("O"."COMPLETED_AT" IS NULL)
    AND ("O"."STATUS" = 'NEW')
    AND (NOT EXISTS(    SELECT
        1
    FROM "RUN"."OUTBOX_RECORD" "EARLIER"
        /* RUN.IDX_OUTBOX_RECORD_KEY_COMPLETED_CREATED_ID: COMPLETED_AT IS NULL
            AND CREATED_AT <= O.CREATED_AT
            AND RECORD_KEY = O.RECORD_KEY
         */
        /* scanCount: 2 */
    WHERE ("EARLIER"."COMPLETED_AT" IS NULL)
        AND ("EARLIER"."CREATED_AT" <= "O"."CREATED_AT")
        AND ("EARLIER"."RECORD_KEY" = "O"."RECORD_KEY")
        AND (("EARLIER"."CREATED_AT" < "O"."CREATED_AT")
        OR (("EARLIER"."CREATED_AT" = "O"."CREATED_AT")
        AND ("EARLIER"."ID" < "O"."ID")))))
ORDER BY 1
FETCH FIRST ROW ONLY
